F) ADD THE OCR JAVASCRIPT HANDLER
Find the end of the existing YOLO `detectForm.addEventListener('submit', ...);` handler.
Immediately AFTER that complete handler and BEFORE the comment `// Native canvas bar charts: no third-party chart CDN is required.`, insert:

  // OCR runs separately from image processing and YOLO.
  ocrForm.addEventListener('submit', async event => {
    event.preventDefault();
    if (!jobId || busy) return;
    busy = true;
    $('processButton').disabled = true;
    $('detectButton').disabled = true;
    $('ocrButton').disabled = true;
    $('ocrDownloadVideo').classList.add('hide');
    $('ocrDownloadCsv').classList.add('hide');
    $('ocrDownloadTxt').classList.add('hide');
    $('ocrSummary').textContent = 'OCR is scanning selected frames. Longer videos can take a while; keep this tab open.';
    message('ocrMessage', 'Starting local Tesseract OCR…');
    status('OCR extraction running');
    try {
      const response = await fetch(`/video/ocr/${encodeURIComponent(jobId)}`, {
        method: 'POST', body: new FormData(ocrForm)
      });
      const result = await response.json();
      if (!response.ok || !result.ok) throw new Error(result.error || 'OCR extraction failed.');

      ocrVideo.src = result.video_url + (result.video_url.includes('?') ? '&' : '?') + 'v=' + Date.now();
      ocrVideo.load();
      setVideoVisible(ocrVideo, 'ocrEmpty', true);
      $('ocrDownloadVideo').href = result.download_url;
      $('ocrDownloadVideo').classList.remove('hide');
      $('ocrDownloadCsv').href = result.csv_url;
      $('ocrDownloadCsv').classList.remove('hide');
      $('ocrDownloadTxt').href = result.text_url;
      $('ocrDownloadTxt').classList.remove('hide');

      const summary = result.summary || {};
      $('ocrSummary').textContent =
        `Frames in output: ${summary.frames_processed ?? '—'} | Frames OCR-scanned: ${summary.frames_ocr_scanned ?? '—'} | Text lines found: ${summary.text_lines_found ?? 0}\n` +
        `Language: ${summary.language || '—'} | Frame interval: ${summary.frame_stride ?? '—'} | Elapsed: ${summary.elapsed_seconds ?? '—'} s`;
      $('ocrVideoMeta').textContent = `${summary.text_lines_found ?? 0} text lines · ${summary.frames_ocr_scanned ?? '—'} sampled frames`;
      message('ocrMessage', 'OCR completed. The annotated video, CSV and timestamped text file are ready.', 'good');
      status('OCR extraction complete');
    } catch (error) {
      message('ocrMessage', error.message, 'error');
      $('ocrSummary').textContent = 'No completed OCR output is available.';
      status('OCR extraction failed', 'error');
    } finally {
      busy = false;
      enableAfterUpload(Boolean(jobId));
    }
  });
